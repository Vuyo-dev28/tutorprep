# Copy this file to .env and fill in your values

# Supabase Configuration
# Get these from your Supabase project settings: https://app.supabase.com/project/_/settings/api
VITE_SUPABASE_URL=https://your-project.supabase.co
VITE_SUPABASE_ANON_KEY=your-anon-key-here

# Application URL (for production)
# Set this to your production domain (e.g., https://tutorprep.co.za)
# Leave empty for development (will use window.location.origin)
# This is used for password reset emails and other redirects
# DO NOT include a trailing slash
VITE_APP_URL=

# Google Generative AI (for content generation)
# Get this from: https://makersuite.google.com/app/apikey
VITE_GOOGLE_GENERATIVE_AI_API_KEY=your-api-key-here

# Paystack Configuration
# Get these from your Paystack dashboard: https://dashboard.paystack.com/#/settings/developer
# Public key (safe to expose in frontend)
VITE_PAYSTACK_PUBLIC_KEY=pk_test_your-public-key-here

# Paystack Secret Key (for subscription initialization)
# ⚠️ WARNING: This exposes your secret key in the frontend bundle.
# For production, consider using a backend API to keep the secret key secure.
VITE_PAYSTACK_SECRET_KEY=sk_test_your-secret-key-here

# Paystack Plan ID (your subscription plan code from Paystack dashboard)
# Create a plan in Paystack Dashboard → Plans, then copy the Plan Code (e.g., PLN_xxxx)
VITE_PAYSTACK_PLAN_ID=PLN_xxxx

# Paystack Plan Amounts (in kobo - e.g., 19900 = 199.00 ZAR)
# These are used for display and can override plan amounts if needed
VITE_PAYSTACK_MONTHLY_AMOUNT=19900
VITE_PAYSTACK_YEARLY_AMOUNT=199000
